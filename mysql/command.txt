CREATE SCHEMA `javase_8th` DEFAULT CHARACTER SET DEFAULT ;

CREATE TABLE `javase_8th`.`item` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(255) NULL,
  `category` VARCHAR(45) NULL,
  `price` INT NULL,
  `qty` INT NULL,
  PRIMARY KEY (`id`));

CREATE TABLE `javase_8th`.`book` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `isbn` VARCHAR(255) NULL,
  `title` VARCHAR(255) NULL,
  `year` INT NULL,
  `publisher_id` INT NULL,
  `category` VARCHAR(255) NULL,
  PRIMARY KEY (`id`));

author 
    id,name,contact

CREATE TABLE `javase_8th`.`author` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(255) NULL,
  `contact` VARCHAR(255) NULL,
  PRIMARY KEY (`id`));
